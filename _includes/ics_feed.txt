{%- comment -%}
DTSTAMP must be in UTC, which Liquid cannot format directly. Shifting the epoch
by the local offset makes the local-time formatter print the UTC clock time.
{%- endcomment -%}
{%- assign ics_offset_hours = site.time | date: "%z" | divided_by: 100 -%}
{%- assign ics_shift = ics_offset_hours | times: 3600 -%}
{%- assign ics_stamp = site.time | date: "%s" | minus: ics_shift | date: "%Y%m%dT%H%M%SZ" -%}

{%- comment -%}
The calendar format requires CRLF line endings and lines of at most 75 bytes.
Long lines continue on the next line after a leading space. Chunks of 60
characters leave room for accented characters, which take two bytes.
{%- endcomment -%}
{%- assign ics_lines = ics_body | split: ics_lf -%}
{%- for ics_raw_line in ics_lines -%}
  {%- assign ics_line = ics_raw_line | strip -%}
  {%- unless ics_line == "" -%}
    {{- ics_line | slice: 0, 60 -}}{{- site.ics_newline -}}
    {%- assign ics_extra = ics_line | size | minus: 1 | divided_by: 60 -%}
    {%- if ics_extra > 0 -%}
      {%- for ics_i in (1..ics_extra) -%}
        {%- assign ics_from = ics_i | times: 60 -%}
        {{- " " -}}{{- ics_line | slice: ics_from, 60 -}}{{- site.ics_newline -}}
      {%- endfor -%}
    {%- endif -%}
  {%- endunless -%}
{%- endfor -%}
